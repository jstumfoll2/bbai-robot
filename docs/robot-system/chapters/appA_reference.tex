\section{SYMBOLS}

\begin{table}[htbp]
\centering
\caption{Main symbols}
\small
\begin{tabular}{cl}
\toprule
$x,y,\psi$ & planar pose; $\psi$ is ENU yaw, counter-clockwise from east \\
$v,\omega$ & body speed and yaw rate \\
$r,b,L$ & wheel radius, track, wheelbase \\
$\chi$ & skid-steer effective-track factor \\
$\delta$ & wheel travel per encoder count \\
$g_z,\ \hat b_g$ & gyroscope $z$ rate and its bias estimate \\
$\bm m,\ \bm o$ & magnetometer reading and hard-iron offset \\
$D$ & magnetic declination, east positive \\
$\chi_c$ & compass heading, clockwise from north ($=\pi/2-\psi$) \\
$\varphi,\lambda,h$ & latitude, longitude, ellipsoidal height \\
$P,Q,R,K,H$ & EKF covariance, process noise, measurement noise, gain, measurement matrix \\
$\ell_i$ & occupancy log-odds of cell $i$ \\
$M,\ N_{\mathrm{eff}}$ & particle count and effective sample size \\
\bottomrule
\end{tabular}
\end{table}

\section{BUILDING THIS DOCUMENT}

cd docs/robot-system

\end{lstlisting}
It needs a TeX distribution with the \texttt{amsmath}, \texttt{siunitx},
\texttt{booktabs}, \texttt{tikz} and \texttt{listings} packages (on Ubuntu:
\texttt{texlive-latex-extra texlive-science}).
